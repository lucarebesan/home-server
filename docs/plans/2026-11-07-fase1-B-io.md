```python
"""Accesso REST a Home Assistant (una richiesta template per tutti gli stati)."""
import json
import urllib.request


def build_template(entities: dict) -> str:
    parts = ", ".join(f"'{k}': states('{v}')" for k, v in entities.items())
    return "{{ {" + parts + "} | tojson }}"


def _num(v):
    try:
        return float(v)
    except (TypeError, ValueError):
        return None


def parse_snapshot(raw: dict, numeric: list[str]) -> dict:
    return {k: (_num(v) if k in numeric else v) for k, v in raw.items()}


class HAApi:
    def __init__(self, base: str, token_file: str, timeout: float = 15):
        self.base = base.rstrip("/")
        with open(token_file) as f:
            self.token = f.read().strip()
        self.timeout = timeout

    def _req(self, path: str, body: dict | None):
        req = urllib.request.Request(
            self.base + path, method="POST" if body is not None else "GET",
            data=json.dumps(body).encode() if body is not None else None,
            headers={"Authorization": f"Bearer {self.token}", "Content-Type": "application/json"})
        with urllib.request.urlopen(req, timeout=self.timeout) as r:
            return r.read().decode()

    def snapshot(self, entities: dict, numeric: list[str]) -> dict:
        txt = self._req("/api/template", {"template": build_template(entities)})
        return parse_snapshot(json.loads(txt), numeric)

    def call_service(self, domain: str, service: str, data: dict) -> None:
        self._req(f"/api/services/{domain}/{service}", data)
```

```python
"""Entità MQTT con discovery automatica per Home Assistant."""
import json
import logging

import paho.mqtt.client as mqtt

log = logging.getLogger(__name__)
AVAIL = "ecm_engine/availability"
DEVICE = {"identifiers": ["ecm_engine"], "name": "ECM Engine", "model": "fase 1", "manufacturer": "Misc"}

SENSORS = [
    ("stato", "Stato motore", "mdi:engine", None),
    ("in_breve", "In breve", "mdi:text", None),
    ("migliore_caldo", "Migliore per scaldare", "mdi:fire", None),
    ("migliore_freddo", "Migliore per raffrescare", "mdi:snowflake", None),
    ("apprendimento_caldo", "Apprendimento caldo", "mdi:school", "%"),
    ("apprendimento_freddo", "Apprendimento freddo", "mdi:school", "%"),
    ("prova_proposta", "Prova proposta", "mdi:flask-outline", None),
    ("prova_in_corso", "Prova in corso", "mdi:flask", None),
    ("dettaglio", "Dettaglio combinazioni", "mdi:table", None),
    ("avvisi", "Avvisi", "mdi:alert", None),
]


def _base(kind, key, name, icon):
    return {"name": name, "unique_id": f"ecm_engine_{key}", "default_entity_id": f"{kind}.ecm_engine_{key}",
            "icon": icon, "availability_topic": AVAIL, "device": DEVICE}


def sensor_config(key, name, icon, unit=None):
    p = _base("sensor", key, name, icon)
    p.update(state_topic=f"ecm_engine/{key}/state", json_attributes_topic=f"ecm_engine/{key}/attr")
    if unit:
        p["unit_of_measurement"] = unit
    return f"homeassistant/sensor/ecm_engine/{key}/config", json.dumps(p)


def button_config(key, name, icon):
    p = _base("button", key, name, icon)
    p["command_topic"] = f"ecm_engine/{key}/cmd"
    return f"homeassistant/button/ecm_engine/{key}/config", json.dumps(p)


def switch_config(key, name, icon):
    p = _base("switch", key, name, icon)
    p.update(command_topic=f"ecm_engine/{key}/set", state_topic=f"ecm_engine/{key}/state")
    return f"homeassistant/switch/ecm_engine/{key}/config", json.dumps(p)


class Publisher:
    def __init__(self, host, port, user=None, password=None, on_command=None):
        self.c = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2, client_id="ecm_engine")
        if user:
            self.c.username_pw_set(user, password)
        self.c.will_set(AVAIL, "offline", retain=True)
        self.on_command = on_command
        self.c.on_connect = self._on_connect
        self.c.on_message = self._on_message
        self.c.connect_async(host, port, keepalive=60)
        self.c.loop_start()

    def _on_connect(self, client, userdata, flags, rc, props=None):
        log.info("MQTT connesso rc=%s", rc)
        for key, name, icon, unit in SENSORS:
            client.publish(*sensor_config(key, name, icon, unit), retain=True)
        client.publish(*button_config("avvia_prova", "Avvia prova proposta", "mdi:play"), retain=True)
        client.publish(*switch_config("prove_automatiche", "Prove automatiche", "mdi:robot"), retain=True)
        client.publish(AVAIL, "online", retain=True)
        client.subscribe("ecm_engine/+/cmd")
        client.subscribe("ecm_engine/+/set")

    def _on_message(self, client, userdata, msg):
        if self.on_command:
            self.on_command(msg.topic, msg.payload.decode())

    def set_state(self, key: str, state, attrs: dict | None = None) -> None:
        self.c.publish(f"ecm_engine/{key}/state", str(state)[:255], retain=True)
        if attrs is not None:
            self.c.publish(f"ecm_engine/{key}/attr", json.dumps(attrs, ensure_ascii=False), retain=True)
```

### Task B3: engine (collector + tester + learning loop)

**Files:** create `ecm_engine/engine.py`, `tests/test_engine.py`

Contract:
- `Engine(api, store, pub, clock=time.time)`.
- `tick_sample()` reads snapshot, builds sample, saves it, updates the warnings, runs the test supervision and the automatic start.
- `tick_learn()` recomputes periods/stats from the last 400 days of samples and publishes everything.
- `on_command(topic, payload)` handles button and switch.
- A test runs as: `call_service("input_number","set_value",{"entity_id":"input_number.ecm_manuale_durata_minuti","value":d})`, then `call_service("input_select","select_option",{"entity_id":"input_select.ecm_manuale","option":label_09(code)})`. Abort selects "AUTO" (except "interrotta dall'utente"). Normal end: when `manuale == "AUTO"` after at least `durata` minutes → esito "completata".

```python
"""Ciclo del motore: campionamento, apprendimento, pubblicazione, prove."""
import logging
import time
from datetime import datetime, time as dtime

from . import config
from .combo import classify, descrizione, label_09, tipo_of, vent_idx
from .learner import aggregate, fascia, learning_pct
from .rules import Ctx, abort_reason, codes_for_season, is_night, propose, start_blockers
from .segmenter import segment
from .texts import best_line, detail_rows, summary_lines

log = logging.getLogger(__name__)
REQUIRED = ["p_cp", "p_tosh", "t_zg", "t_zn", "t_ext"]
SAMPLE_FIELDS = ["tosh_rps", "p_cp", "p_tosh", "t_zg", "t_zn", "t10", "t2", "t_ext", "soc", "pv",
                 "load", "imp", "discharge", "charge"]


def build_sample(snap: dict, ts: int, open_test: dict | None) -> dict:
    c = classify(snap)
    s = {k: snap.get(k) for k in SAMPLE_FIELDS}
    s.update(ts=ts, stagione=snap.get("stagione"), test_id=None,
             cp_compressore=1 if str(snap.get("cp_compressore")).upper() == "ON" else 0,
             code=c.code, tipo=c.tipo, valido=1, motivo=c.motivo)
    if c.code is None:
        s["valido"] = 0
    missing = [k for k in REQUIRED if snap.get(k) is None]
    if s["valido"] and missing:
        s["valido"], s["motivo"] = 0, f"dato mancante {missing[0]}"
    if open_test:
        s["test_id"] = open_test["id"]
        tc = open_test["code"]
        if s["valido"]:
            if c.code // 10 == tc // 10 and vent_idx(c.code % 10) == vent_idx(tc % 10):
                s["code"], s["tipo"] = tc, tipo_of(tc, snap.get("stagione"))
            else:
                s["valido"], s["motivo"] = 0, "prova non ancora applicata"
    return s


def _t(v, default):
    try:
        h, m = str(v).split(":")[:2]
        return dtime(int(h), int(m))
    except (ValueError, TypeError):
        return default


class Engine:
    def __init__(self, api, store, pub, clock=time.time):
        self.api, self.store, self.pub, self.clock = api, store, pub, clock
        self.snap: dict | None = None
        self.stats: dict = {}
        self.proposal = None
        self.missing_since: dict[str, float] = {}
        self.ha_error: str | None = None

    # ---------- contesto
    def _ctx(self) -> Ctx | None:
        s = self.snap
        if not s or any(s.get(k) is None for k in ("t_zg", "t_zn", "lim_inf", "lim_sup", "soc")):
            return None
        return Ctx(now=datetime.fromtimestamp(self.clock()), stagione=s["stagione"],
                   manuale=s["manuale"], free_cooling=s["free_cooling"],
                   acs_attivo=s.get("acs_inv") == "on" or s.get("acs_soc") == "on",
                   soc=s["soc"], pv=s.get("pv") or 0.0, load=s.get("load") or 0.0,
                   t_zg=s["t_zg"], t_zn=s["t_zn"], lim_inf=s["lim_inf"], lim_sup=s["lim_sup"],
                   mattina=_t(s.get("mattina"), dtime(8, 30)), serale=_t(s.get("serale"), dtime(22, 0)),
                   t_ext=s.get("t_ext") if s.get("t_ext") is not None else 10.0)

    def _day_start(self) -> float:
        d = datetime.fromtimestamp(self.clock())
        return d.replace(hour=0, minute=0, second=0, microsecond=0).timestamp()

    # ---------- campionamento
    def tick_sample(self) -> None:
        now = int(self.clock())
        try:
            self.snap = self.api.snapshot(config.ENTITIES, config.NUMERIC_KEYS)
            self.ha_error = None
        except Exception as ex:  # rete o HA giù
            self.ha_error = f"HA non raggiungibile: {ex}"
            log.warning(self.ha_error)
            self._publish_status()
            return
        sample = build_sample(self.snap, now, self.store.open_test())
        self.store.add_sample(sample)
        for k in REQUIRED:
            if self.snap.get(k) is None:
                self.missing_since.setdefault(k, now)
            else:
                self.missing_since.pop(k, None)
        self._supervise()
        self._auto_start()
        self._publish_status()

    # ---------- prove
    def start_test(self, origine: str) -> str | None:
        c = self._ctx()
        p = self.proposal
        if c is None or p is None:
            return "dati non disponibili"
        last = self.store.last_test_end()
        blockers = start_blockers(c, p.tipo, origine == "auto",
                                  len(self.store.tests_since(self._day_start(), origine="auto")),
                                  datetime.fromtimestamp(last) if last else None)
        if self.store.open_test():
            blockers.append("prova già in corso")
        if blockers:
            return blockers[0]
        self.api.call_service("input_number", "set_value",
                              {"entity_id": "input_number.ecm_manuale_durata_minuti", "value": p.durata})
        self.api.call_service("input_select", "select_option",
                              {"entity_id": "input_select.ecm_manuale", "option": label_09(p.code)})
        self.store.start_test(p.code, p.durata, origine, int(self.clock()))
        log.info("prova %s avviata: E%s per %s min", origine, p.code, p.durata)
        return None

    def _supervise(self) -> None:
        t = self.store.open_test()
        c = self._ctx()
        if not t or c is None:
            return
        now = int(self.clock())
        label = label_09(t["code"])
        elapsed = now - t["inizio"]
        if elapsed < 90 and c.manuale != label:
            return  # 09 non ha ancora applicato la selezione
        if c.manuale == "AUTO" and elapsed >= t["durata_min"] * 60 - 60:
            self.store.end_test(t["id"], now, "completata")
            return
        reason = abort_reason(c, tipo_of(t["code"], c.stagione), label)
        if reason is None:
            return
        if reason != "interrotta dall'utente":
            self.api.call_service("input_select", "select_option",
                                  {"entity_id": "input_select.ecm_manuale", "option": "AUTO"})
        self.store.end_test(t["id"], now, reason)
        log.info("prova %s chiusa: %s", t["id"], reason)

    def _auto_start(self) -> None:
        if self.store.get_setting("auto", "0") != "1" or self.store.open_test():
            return
        self.start_test("auto")

    def recover(self) -> None:
        """All'avvio: chiude una prova rimasta aperta."""
        t = self.store.open_test()
        if not t:
            return
        try:
            snap = self.api.snapshot(config.ENTITIES, config.NUMERIC_KEYS)
            if snap.get("manuale") == label_09(t["code"]):
                self.api.call_service("input_select", "select_option",
                                      {"entity_id": "input_select.ecm_manuale", "option": "AUTO"})
        except Exception as ex:
            log.warning("recover: %s", ex)
        self.store.end_test(t["id"], int(self.clock()), "interrotta dal riavvio del motore")

    def on_command(self, topic: str, payload: str) -> None:
        if topic == "ecm_engine/avvia_prova/cmd":
            err = self.start_test("manuale")
            if err:
                log.info("avvio manuale rifiutato: %s", err)
        elif topic == "ecm_engine/prove_automatiche/set":
            on = payload.strip().upper() == "ON"
            self.store.set_setting("auto", "1" if on else "0")
            self.pub.set_state("prove_automatiche", "ON" if on else "OFF")
        self._publish_status()

    # ---------- apprendimento e pubblicazione
    def tick_learn(self) -> None:
        now = int(self.clock())
        self.store.purge(now - config.RETENTION_DAYS * 86400)
        periods = segment(self.store.samples_since(now - config.RETENTION_DAYS * 86400))
        self.stats = aggregate(periods)
        c = self._ctx()
        self.proposal = propose(self.stats, c) if c else None
        self._publish_learn()
        self._publish_status()

    def _publish_learn(self) -> None:
        c = self._ctx()
        if c is None:
            return
        f = fascia(c.t_ext)
        night = is_night(c)
        heat = [x for x in codes_for_season("INVERNO", night) if tipo_of(x, "INVERNO") == "HEAT"]
        cool = [x for x in codes_for_season("ESTATE", night) if tipo_of(x, "ESTATE") in ("COOL", "DRY")]
        pct_h, pct_c = learning_pct(self.stats, heat, f), learning_pct(self.stats, cool, f)
        pct = pct_h if c.stagione == "INVERNO" else pct_c
        lines = summary_lines(self.stats, c.stagione, f, pct)
        self.pub.set_state("in_breve", lines[0], {"righe": lines, "fascia_esterna": f})
        self.pub.set_state("migliore_caldo", best_line(self.stats, "HEAT", f, c.stagione))
        self.pub.set_state("migliore_freddo", best_line(self.stats, "COOL", f, c.stagione))
        self.pub.set_state("apprendimento_caldo", pct_h)
        self.pub.set_state("apprendimento_freddo", pct_c)
        rows = detail_rows(self.stats, f, c.stagione)
        self.pub.set_state("dettaglio", len(rows), {"fascia_esterna": f, "tabella": rows})

    def _publish_status(self) -> None:
        now = int(self.clock())
        avvisi = [f"{k} non disponibile da {round((now - t0) / 60)} minuti"
                  for k, t0 in self.missing_since.items() if now - t0 >= 1800]
        if self.ha_error:
            avvisi.append(self.ha_error)
        self.pub.set_state("avvisi", len(avvisi), {"elenco": avvisi})
        t = self.store.open_test()
        if t:
            stagione = (self.snap or {}).get("stagione", "INVERNO")
            self.pub.set_state("prova_in_corso", descrizione(t["code"], stagione),
                               {"codice": f"E{t['code']}", "origine": t["origine"],
                                "inizio": datetime.fromtimestamp(t["inizio"]).isoformat(timespec="minutes"),
                                "fine_prevista": datetime.fromtimestamp(t["inizio"] + t["durata_min"] * 60).isoformat(timespec="minutes"),
                                "minuti_trascorsi": round((now - t["inizio"]) / 60)})
            stato = "prova in corso"
        else:
            self.pub.set_state("prova_in_corso", "nessuna", {})
            stato = "errore HA" if self.ha_error else "online"
        self.pub.set_state("stato", stato)
        self.pub.set_state("prove_automatiche", "ON" if self.store.get_setting("auto", "0") == "1" else "OFF")
        c, p = self._ctx(), self.proposal
        if p and c:
            last = self.store.last_test_end()
            blk = start_blockers(c, p.tipo, False, 0, datetime.fromtimestamp(last) if last else None)
            self.pub.set_state("prova_proposta", f"{descrizione(p.code, c.stagione)} per {p.durata} minuti",
                               {"codice": f"E{p.code}", "durata": p.durata, "motivo": p.motivo,
                                "avviabile": not blk and not t, "motivo_blocco": blk[0] if blk else ("prova in corso" if t else "")})
        else:
            self.pub.set_state("prova_proposta", "nessuna", {"avviabile": False, "motivo_blocco": "dati non disponibili"})
```

```python
def main():
    store = Store(config.DB_PATH)
    api = HAApi(config.HA_BASE, config.HA_TOKEN_FILE)
    holder = {}
    pub = Publisher(config.MQTT_HOST, config.MQTT_PORT, os.environ.get("MQTT_USER"),
                    os.environ.get("MQTT_PASS"), on_command=lambda t, p: holder["e"].on_command(t, p))
    engine = Engine(api, store, pub)
    holder["e"] = engine
    engine.recover()
    next_learn = 0.0
    while True:
        t0 = time.time()
        try:
            engine.tick_sample()
            if t0 >= next_learn:
                engine.tick_learn()
                next_learn = t0 + config.LEARN_PERIOD_S
        except Exception:
            log.exception("errore nel ciclo")
        time.sleep(max(1.0, config.SAMPLE_PERIOD_S - (time.time() - t0)))


if __name__ == "__main__":
    main()
```
