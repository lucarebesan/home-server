```python
"""Stato reale delle macchine -> codice combinazione E (numerazione HA)."""
from dataclasses import dataclass

CP_VENT = {"OFF": 0, "STEP 2": 2, "STEP 3": 3, "STEP 4": 4}
TOSH_VENT = {"BASSA": 1, "MEDIA": 2, "ALTA": 3}
TOSH_MODO_BASE = {"DEUMIDIFICATORE": 0, "RAFFRESCAMENTO": 3, "RISCALDAMENTO": 6}
CP_STATI_ESCLUSI = {"ACS", "ANTILEGIONELLA", "SBRINAMENTO", "ALLARME", "SERVICE", "PROTEZIONE ANTIGELO"}
SPEED_NAME = {1: "bassa", 2: "media", 3: "alta"}


@dataclass(frozen=True)
class Combo:
    code: int | None
    tipo: str | None
    motivo: str | None


def _u(v):
    return (v or "").strip().upper()


def vent_idx(u: int) -> int:
    """0 = V0, 1/2/3 = V2/V3/V4 indipendentemente dal compressore."""
    return 0 if u == 0 else (u - 1) % 3 + 1


def cp_unit(cp_vent: str, cp_compressore: str) -> int | None:
    v = CP_VENT.get(_u(cp_vent))
    if v is None:
        return None
    on = _u(cp_compressore) == "ON"
    if v == 0:
        return None if on else 0
    return (v - 1) + (3 if on else 0)


def tosh_decade(on: str, modo: str, vent: str) -> int | None:
    if _u(on) != "ACCESO":
        return 0
    base = TOSH_MODO_BASE.get(_u(modo))
    v = TOSH_VENT.get(_u(vent))
    if base is None or v is None:
        return None
    return base + v


def tipo_of(code: int, stagione: str) -> str:
    d, u = divmod(code, 10)
    if d >= 7:
        return "HEAT"
    if d >= 4:
        return "COOL"
    if d >= 1:
        return "DRY"
    if u >= 4:
        return "HEAT" if _u(stagione) == "INVERNO" else "COOL"
    return "NEUTRO"


def classify(s: dict) -> Combo:
    stato = _u(s.get("cp_stato"))
    if stato in CP_STATI_ESCLUSI:
        return Combo(None, None, f"cp_stato {stato}")
    if _u(s.get("cp_resistenza")) == "ON":
        return Combo(None, None, "resistenza ACS")
    if _u(s.get("free_cooling")) in ("ATTIVATO", "PREPARAZIONE"):
        return Combo(None, None, "free cooling")
    u = cp_unit(s.get("cp_vent"), s.get("cp_compressore"))
    if u is None:
        return Combo(None, None, "cp non classificato")
    d = tosh_decade(s.get("tosh_on"), s.get("tosh_modo"), s.get("tosh_vent"))
    if d is None:
        return Combo(None, None, "toshiba non classificato")
    code = 10 * d + u
    return Combo(code, tipo_of(code, s.get("stagione")), None)


def _cp_label(u: int) -> str:
    if u == 0:
        return "CP OFF V0"
    return f"CP {'ON' if u >= 4 else 'OFF'} V{vent_idx(u) + 1}"


def _split_label(d: int) -> str:
    if d == 0:
        return "Split OFF"
    mode = ["DRY", "COOL", "HEAT"][(d - 1) // 3]
    return f"{mode} V{(d - 1) % 3 + 1}"


def label_09(code: int) -> str:
    d, u = divmod(code, 10)
    return f"E{code} - {_cp_label(u)} • {_split_label(d)}"


def descrizione(code: int, stagione: str) -> str:
    d, u = divmod(code, 10)
    if u == 0:
        cp = "Nilan ferma"
    elif u >= 4:
        cp = f"Nilan pompa di calore V{vent_idx(u) + 1}"
    else:
        cp = f"Nilan solo ventilazione V{vent_idx(u) + 1}"
    if d == 0:
        t = "Toshiba spento"
    else:
        mode = ["deumidifica", "freddo", "caldo"][(d - 1) // 3]
        t = f"Toshiba {mode} {SPEED_NAME[(d - 1) % 3 + 1]}"
    return f"{cp} + {t}"
```

```python
"""Persistenza SQLite: campioni, prove, impostazioni."""
import sqlite3

SAMPLE_COLS = ["ts", "code", "tipo", "valido", "motivo", "stagione", "test_id", "cp_compressore",
               "tosh_rps", "p_cp", "p_tosh", "t_zg", "t_zn", "t10", "t2", "t_ext", "soc", "pv",
               "load", "imp", "discharge", "charge"]

SCHEMA = """
CREATE TABLE IF NOT EXISTS samples (
  ts INTEGER PRIMARY KEY, code INTEGER, tipo TEXT, valido INTEGER, motivo TEXT, stagione TEXT,
  test_id INTEGER, cp_compressore INTEGER, tosh_rps REAL, p_cp REAL, p_tosh REAL, t_zg REAL,
  t_zn REAL, t10 REAL, t2 REAL, t_ext REAL, soc REAL, pv REAL, load REAL, imp REAL,
  discharge REAL, charge REAL);
CREATE TABLE IF NOT EXISTS tests (
  id INTEGER PRIMARY KEY AUTOINCREMENT, code INTEGER, durata_min INTEGER, origine TEXT,
  inizio INTEGER, fine INTEGER, esito TEXT);
CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT);
"""


class Store:
    def __init__(self, path: str):
        self.db = sqlite3.connect(path, check_same_thread=False)
        self.db.row_factory = sqlite3.Row
        self.db.executescript(SCHEMA)
        self.db.commit()

    def add_sample(self, s: dict) -> None:
        vals = [s.get(c) for c in SAMPLE_COLS]
        self.db.execute(f"INSERT OR REPLACE INTO samples ({','.join(SAMPLE_COLS)}) "
                        f"VALUES ({','.join('?' * len(SAMPLE_COLS))})", vals)
        self.db.commit()

    def samples_since(self, ts: int) -> list[dict]:
        cur = self.db.execute("SELECT * FROM samples WHERE ts >= ? ORDER BY ts", (ts,))
        return [dict(r) for r in cur]

    def purge(self, older_than_ts: int) -> None:
        self.db.execute("DELETE FROM samples WHERE ts < ?", (older_than_ts,))
        self.db.commit()

    def start_test(self, code: int, durata_min: int, origine: str, ts: int) -> int:
        cur = self.db.execute("INSERT INTO tests (code, durata_min, origine, inizio) VALUES (?,?,?,?)",
                              (code, durata_min, origine, ts))
        self.db.commit()
        return cur.lastrowid

    def end_test(self, test_id: int, ts: int, esito: str) -> None:
        self.db.execute("UPDATE tests SET fine=?, esito=? WHERE id=?", (ts, esito, test_id))
        self.db.commit()

    def open_test(self) -> dict | None:
        r = self.db.execute("SELECT * FROM tests WHERE fine IS NULL ORDER BY id DESC LIMIT 1").fetchone()
        return dict(r) if r else None

    def last_test_end(self) -> int | None:
        r = self.db.execute("SELECT MAX(fine) FROM tests").fetchone()
        return r[0]

    def tests_since(self, ts: int, origine: str | None = None) -> list[dict]:
        q, a = "SELECT * FROM tests WHERE inizio >= ?", [ts]
        if origine:
            q += " AND origine = ?"; a.append(origine)
        return [dict(r) for r in self.db.execute(q, a)]

    def get_setting(self, key: str, default: str) -> str:
        r = self.db.execute("SELECT value FROM settings WHERE key=?", (key,)).fetchone()
        return r[0] if r else default

    def set_setting(self, key: str, value: str) -> None:
        self.db.execute("INSERT OR REPLACE INTO settings (key, value) VALUES (?,?)", (key, value))
        self.db.commit()
```

```python
"""Campioni -> periodi a combinazione costante."""
from dataclasses import dataclass


@dataclass
class Period:
    code: int
    tipo: str
    start: int
    end: int
    minutes: int
    p_cp: float
    p_tosh: float
    p_tot: float
    duty_cp: float
    duty_tosh: float
    slope_zg: float
    slope_zn: float
    t_ext: float
    da_prova: bool


def slope_per_hour(ts: list[int], ys: list[float]) -> float:
    n = len(ts)
    if n < 2:
        return 0.0
    xs = [t / 3600 for t in ts]
    mx, my = sum(xs) / n, sum(ys) / n
    den = sum((x - mx) ** 2 for x in xs)
    if den == 0:
        return 0.0
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den


def _mean(v):
    return sum(v) / len(v) if v else 0.0


def _close(group, settle_s, min_useful_s):
    t0 = group[0]["ts"]
    rows = [r for r in group if r["ts"] >= t0 + settle_s]
    if not rows or (rows[-1]["ts"] - rows[0]["ts"] + 60) < min_useful_s:
        return None
    ts = [r["ts"] for r in rows]
    p_cp = _mean([r["p_cp"] for r in rows])
    p_tosh = _mean([r["p_tosh"] for r in rows])
    return Period(
        code=rows[0]["code"], tipo=rows[0]["tipo"], start=ts[0], end=ts[-1], minutes=len(rows),
        p_cp=round(p_cp, 1), p_tosh=round(p_tosh, 1), p_tot=round(p_cp + p_tosh, 1),
        duty_cp=_mean([1.0 if r["cp_compressore"] else 0.0 for r in rows]),
        duty_tosh=_mean([1.0 if (r["tosh_rps"] or 0) > 0 else 0.0 for r in rows]),
        slope_zg=slope_per_hour(ts, [r["t_zg"] for r in rows]),
        slope_zn=slope_per_hour(ts, [r["t_zn"] for r in rows]),
        t_ext=_mean([r["t_ext"] for r in rows]),
        da_prova=rows[0]["test_id"] is not None,
    )


def segment(samples: list[dict], gap_s: int = 180, settle_s: int = 300,
            min_useful_s: int = 600) -> list[Period]:
    out, group = [], []
    for r in samples:
        ok = r.get("valido") and r.get("code") is not None
        if group and ok and r["code"] == group[-1]["code"] \
                and r.get("test_id") == group[-1].get("test_id") \
```

```python
"""Aggregazione periodi -> statistiche per (combinazione, fascia esterna)."""
from dataclasses import dataclass
from statistics import pstdev

FASCE = [(-99, 5, "<5"), (5, 10, "5-10"), (10, 15, "10-15"), (15, 20, "15-20"),
         (20, 25, "20-25"), (25, 30, "25-30"), (30, 99, "≥30")]
NOMI = [f[2] for f in FASCE]
SIGN = {"HEAT": 1, "COOL": -1, "DRY": -1, "NEUTRO": 0}
EFF_MIN = 0.1


@dataclass
class Stat:
    code: int
    tipo: str
    fascia: str
    periodi: int
    minuti: int
    p_tot: float
    duty_cp: float
    duty_tosh: float
    eff_zg: float
    eff_zn: float
    eff: float
    costo: float | None
    affidabilita: float


def fascia(t_ext: float) -> str:
    for lo, hi, name in FASCE:
        if lo <= t_ext < hi:
            return name
    return NOMI[-1]


def _wmean(vals, weights):
    tw = sum(weights)
    return sum(v * w for v, w in zip(vals, weights)) / tw if tw else 0.0


def aggregate(periods) -> dict:
    groups = {}
    for p in periods:
        groups.setdefault((p.code, fascia(p.t_ext)), []).append(p)
    out = {}
    for (code, f), ps in groups.items():
        tipo, sign = ps[0].tipo, SIGN[ps[0].tipo]
        w = [p.minutes for p in ps]
        eff_zg = sign * _wmean([p.slope_zg for p in ps], w)
        eff_zn = sign * _wmean([p.slope_zn for p in ps], w)
        eff = (eff_zg + eff_zn) / 2
        p_tot = _wmean([p.p_tot for p in ps], w)
        if sign == 0:
            serie = [p.p_tot for p in ps]
        else:
            serie = [sign * (p.slope_zg + p.slope_zn) / 2 for p in ps]
        m = sum(serie) / len(serie)
        coerente = len(serie) < 2 or pstdev(serie) <= 0.5 * abs(m)
        aff = min(len(ps) / 5, 1.0) * (1.0 if coerente else 0.6)
        costo = p_tot / eff if sign != 0 and eff > EFF_MIN else None
        out[(code, f)] = Stat(code, tipo, f, len(ps), sum(w), round(p_tot, 1),
                              _wmean([p.duty_cp for p in ps], w), _wmean([p.duty_tosh for p in ps], w),
                              eff_zg, eff_zn, eff, costo, aff)
    return out


def _vicine(f: str) -> list[str]:
    i = NOMI.index(f)
    return [NOMI[j] for j in (i - 1, i + 1) if 0 <= j < len(NOMI)]


def ranking(stats: dict, tipo: str, f: str) -> list[Stat]:
    sel = [s for s in stats.values() if s.tipo == tipo and s.fascia == f]
    if not sel:
        sel = [s for s in stats.values() if s.tipo == tipo and s.fascia in _vicine(f)]
    return sorted(sel, key=lambda s: (s.costo is None, s.costo or 0, s.p_tot))


def learning_pct(stats: dict, codes: list[int], f: str) -> int:
    if not codes:
        return 0
    tot = sum(stats[(c, f)].affidabilita if (c, f) in stats else 0.0 for c in codes)
    return round(100 * tot / len(codes))
```

```python
"""Candidati, proposta di prova, condizioni di avvio e di interruzione."""
from dataclasses import dataclass
from datetime import datetime, time, timedelta

from .combo import tipo_of
from .learner import EFF_MIN, fascia

AUTO_WINDOW = (time(10, 0), time(17, 0))
AUTO_MAX_PER_DAY = 3
AUTO_PAUSE = timedelta(minutes=60)
MARGINE = 0.3


@dataclass
class Ctx:
    now: datetime
    stagione: str
    manuale: str
    free_cooling: str
    acs_attivo: bool
    soc: float
    pv: float
    load: float
    t_zg: float
    t_zn: float
    lim_inf: float
    lim_sup: float
    mattina: time
    serale: time
    t_ext: float


@dataclass
class Proposal:
    code: int
    tipo: str
    durata: int
    motivo: str


def is_night(c: Ctx) -> bool:
    t = c.now.time()
    return not (c.mattina <= t < c.serale)


def codes_for_season(stagione: str, night: bool) -> list[int]:
    if stagione == "INVERNO":
        codes = [1, 2, 3, 4, 5, 6] + [10 * d + u for d in (7, 8, 9) for u in range(7)]
    else:
        codes = [1, 2, 3, 4, 5, 6] + [10 * d + u for d in range(1, 7) for u in range(7)]
    if night:
        codes = [c for c in codes if c % 10 not in (3, 6) and not (c >= 10 and (c // 10) % 3 == 0)]
    return codes


def _energia_ok(c: Ctx, soc_min: float) -> bool:
    return c.soc >= soc_min or c.pv >= c.load


def start_blockers(c: Ctx, tipo: str, auto: bool, tests_today: int,
                   last_end: datetime | None) -> list[str]:
    b = []
    if c.manuale != "AUTO":
        b.append("modalità manuale già attiva")
    if c.free_cooling in ("ATTIVATO", "PREPARAZIONE"):
        b.append("free cooling attivo")
    if c.acs_attivo:
        b.append("ciclo ACS in corso")
    if not _energia_ok(c, 40):
        b.append("energia insufficiente")
    zmax, zmin = max(c.t_zg, c.t_zn), min(c.t_zg, c.t_zn)
    lo, hi = c.lim_inf + MARGINE, c.lim_sup - MARGINE
    if tipo in ("HEAT", "NEUTRO") and zmax > hi:
        b.append("zona troppo calda per scaldare" if tipo == "HEAT" else "zona fuori fascia")
    if tipo in ("COOL", "DRY", "NEUTRO") and zmin < lo:
        b.append("zona troppo fredda per raffrescare" if tipo != "NEUTRO" else "zona fuori fascia")
    if auto:
        if not (AUTO_WINDOW[0] <= c.now.time() < AUTO_WINDOW[1]):
            b.append("fuori finestra prove automatiche")
        if tests_today >= AUTO_MAX_PER_DAY:
            b.append(f"già {AUTO_MAX_PER_DAY} prove oggi")
        if last_end and c.now - last_end < AUTO_PAUSE:
            b.append("pausa tra le prove")
    return b


def abort_reason(c: Ctx, tipo: str, expected_label: str) -> str | None:
    if c.manuale != expected_label:
        return "interrotta dall'utente"
    zmax, zmin = max(c.t_zg, c.t_zn), min(c.t_zg, c.t_zn)
    if tipo in ("HEAT", "NEUTRO") and zmax > c.lim_sup:
        return "zona oltre il limite superiore"
    if tipo in ("COOL", "DRY", "NEUTRO") and zmin < c.lim_inf:
        return "zona sotto il limite inferiore"
    if c.soc < 25 and c.pv < c.load:
        return "batteria sotto 25%"
    return None


def propose(stats: dict, c: Ctx) -> Proposal | None:
    f = fascia(c.t_ext)
    cands = codes_for_season(c.stagione, is_night(c))
    if not cands:
        return None

    def key(code):
        st = stats.get((code, f))
        tipo = tipo_of(code, c.stagione)
        return (st.affidabilita if st else 0.0, tipo == "NEUTRO", code)

    code = min(cands, key=key)
    tipo = tipo_of(code, c.stagione)
    st = stats.get((code, f))
    if tipo == "NEUTRO":
        durata, motivo = 15, "manca il consumo"
    elif st is None or abs(st.eff) < EFF_MIN:
        durata, motivo = 30, "manca l'efficacia"
    else:
        durata, motivo = 20, "dato da confermare"
    return Proposal(code, tipo, durata, motivo)
```

```python
"""Frasi semplici per Luca e righe della tabella di dettaglio."""
from .combo import descrizione
from .learner import EFF_MIN, ranking

VERBO = {"HEAT": "scaldare", "COOL": "raffrescare"}


def best_line(stats, tipo, f, stagione) -> str:
    r = [s for s in ranking(stats, tipo, f) if s.costo is not None]
    if not r:
        return f"Per {VERBO[tipo]}: non ho ancora abbastanza dati"
    s = r[0]
    return (f"Per {VERBO[tipo]} adesso conviene: {descrizione(s.code, stagione)} "
            f"(circa {round(s.p_tot)} W)")


def summary_lines(stats, stagione, f, pct) -> list[str]:
    tipo = "HEAT" if stagione == "INVERNO" else "COOL"
    lines = [best_line(stats, tipo, f, stagione)]
    for s in ranking(stats, tipo, f):
        if s.affidabilita >= 0.4 and s.eff < EFF_MIN:
            lines.append(f"{descrizione(s.code, stagione)}: non rende con questo clima esterno")
    lines.append(f"Ho imparato il {pct}% di quello che serve per {VERBO[tipo]}")
    return lines


def detail_rows(stats, f, stagione) -> list[dict]:
    rows = []
    for s in sorted((s for s in stats.values() if s.fascia == f), key=lambda s: s.code):
        rows.append({
            "codice": f"E{s.code}", "descrizione": descrizione(s.code, stagione),
            "W": round(s.p_tot), "duty": round(s.duty_cp, 2),
            "zg": round(s.eff_zg, 2), "zn": round(s.eff_zn, 2),
            "costo": round(s.costo) if s.costo is not None else None,
            "periodi": s.periodi, "affidabilita": round(s.affidabilita, 2),
        })
    return rows
```
